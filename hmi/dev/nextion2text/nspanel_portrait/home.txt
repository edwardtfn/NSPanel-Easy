                swipec2=0-swipedx
                if(swipec<swipec2)
                {
                    // Navigate back based on active status
                    if(is_buttonpage01>=1)
                    {
                        page buttonpage01
                    }else if(is_buttonpage02>=1)
                    {
                        page buttonpage02
                    }else if(is_buttonpage03>=1)

                swipec=swipey-tch3

                if(swipec>100)

                    if(is_buttonpage04>=1&&is_buttonpage03>=1)

                    {
                        page buttonpage01
                    }

                // Down to Up
                swipec2=0-swipedy
                if(swipec<-100)
                {
