Timer init
    Attributes
        ID         : 20
        Scope      : local
        Period (ms): 50
        Enabled    : no
    
    Events
        Timer Event
            // init timer event
            init.en=0
            // Hide everything, so a page change repaints from a clean card
            vis 255,0
            // Card rows: all options up to 8 (keeps full height on a partial last page)
            sys2=total_count.val
            if(sys2>8)
            {
                sys2=8
            }
            if(display_mode==2)
            {
                // Portrait (card 280 wide, starts at 20,40)
                sys0=sys2*39
                sys0+=41
                if(total_count.val>8)
                {
                    sys0+=40  // footer row
                }
                fill 20,40,280,sys0,21197
                sys1=sys0-8
                fill 24,44,272,sys1,61310
            }else
            {
                // Landscape (card 350 wide, starts at 50,20)
                sys0=sys2*28
                sys0+=38
                if(total_count.val>8)
                {
                    sys0+=30  // footer row
                }
                fill 50,20,350,sys0,21197
                sys1=sys0-8
                fill 54,24,342,sys1,61310
            }
            // Show static header components
            vis header_bg,1
            vis btn_back,1
            vis lbl_title,1
            // Show footer (page navigation) only when paged
            if(total_count.val>8)
            {
                vis btn_prev,1
                vis lbl_hint,1
                vis btn_next,1
            }
            // Show populated opt buttons
            sys1=opt_count.val+6
            for(sys0=6;sys0<sys1;sys0++)
            {
                vis b[sys0],1
            }
            // Kick render to paint selection state
            render.en=1
        
Timer tap_eval
    Attributes
        ID         : 25
        Scope      : local
        Period (ms): 50
        Enabled    : no
    
    Events
        Timer Event
            // tap_eval timer event: tells a swipe from a tap on an option button
            tap_eval.en=0
            sys0=press_y.val-tch3  // positive: finger moved up
            sys1=tch3-press_y.val  // positive: finger moved down
            if(sys0>=40)
            {
                // Swipe up: next page, reported as a btn_next (22) release
                printh 65
                prints dp,1
                printh 16 00 FF FF FF
                render.en=1
            }else if(sys1>=40)
            {
                // Swipe down: previous page, reported as a btn_prev (21) release
                printh 65
                prints dp,1
                printh 15 00 FF FF FF
                render.en=1
            }else
            {
                // Tap: local feedback, then report the option button release
                sys0=tap_slot.val
                sys1=sys0+6  // component id
                if(select_mode.val==0)
                {
                    if(b[sys1].val==1)
                    {
                        selection_mask.val=1<<sys0
                    }else
                    {
                        selection_mask.val=0
                    }
                }
                render.en=1
                printh 65
                prints dp,1
                prints sys1,1
                printh 00 FF FF FF
            }
